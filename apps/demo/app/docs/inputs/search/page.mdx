## Search

Search allows the user to quickly find something using a query.

```tsx
<Search hotkey />
```

### Usage

Manage the search query with `useState` and handle changes with `onChange`.

```tsx
const [query, setQuery] = useState("");

return <Search value={query} onChange={setQuery} />;
```

### Props

- [`ElementCustomizableProps`](/docs/guides/shared-props) available: `element`.

<PropsTable>
  <PropsTableRow prop="value" required="never">

    <TableCell>

      The value inside the field. This is useful if you want a controlled input.

      - Optional.

      </TableCell>

  </PropsTableRow>

  <PropsTableRow prop="hotkey" required="never">
    <TableCell>`boolean | string`</TableCell>

      <kbd className="kbd">/</kbd>, or any other key you specify.

      - Must be either boolean or a single character string.
      - Optional.

      </TableCell>

  </PropsTableRow>

  <PropsTableRow prop="locale" required="never">
    <TableCell>`"en-US" | "th"`</TableCell>
    <TableCell>

      Allows for translation of the default placeholder message.

      - Must be `th` or `en-US`, SKCom supports two languages: th and en-US.
      - Optional.

      </TableCell>

  </PropsTableRow>

  <PropsTableRow prop="onChange" required="never">
    <TableCell>`(value: string) => any`</TableCell>
    <TableCell>

      Called when the user makes changes to the field value. The value is passed
      in via the function.

      - Optional.

      </TableCell>

  </PropsTableRow>

  <PropsTableRow prop="onSearch" required="never">
    <TableCell>`(value: string) => any`</TableCell>
    <TableCell>

      Called when the search button is clicked. The value is passed in via the
      function.

      - Optional.

      </TableCell>

  </PropsTableRow>

  <PropsTableRow prop="placeholder" required="never">
    <TableCell>`string`</TableCell>
    <TableCell>

      A faint text displayed inside the field guiding the user.

      - Optional.

      </TableCell>

  </PropsTableRow>

  <PropsTableRow prop="disabled" required="never">
    <TableCell>`boolean`</TableCell>
